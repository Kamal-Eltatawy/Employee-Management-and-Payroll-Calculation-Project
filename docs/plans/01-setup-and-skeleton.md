# Phase 1 — Setup & Skeleton Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: use `superpowers:executing-plans` to implement this plan task by task. Steps use checkbox (`- [ ]`) syntax for tracking. Read `docs/plans/00-master-plan.md` (Global Constraints, Execution model, Shared contracts) first. Every task here implicitly includes them.

**Goal:** A buildable, layered .NET 10 solution with centralized package versions, RFC 7807 error handling (tested), an API host with OpenAPI, a local SQL Server in Docker, and an Angular 22 client with Bootstrap and a shell layout.

**Architecture:** There are 4 projects (Domain → Application → Infrastructure → Api), and project references enforce the dependency direction. All exceptions become Problem Details in one `IExceptionHandler`. The Angular client is a standalone app that proxies `/api` to the backend in development.

**Tech Stack:** .NET 10 SDK, ASP.NET Core, FluentValidation (exception type only, at this stage), xUnit v3, Docker (SQL Server 2022), Angular CLI 22.2.6, Bootstrap 5.3.8.

  'Microsoft.AspNetCore.Mvc.Testing'                                 = '10.0.'
  # Latest stable
  'Microsoft.Extensions.TimeProvider.Testing'      = ''
  'Microsoft.IdentityModel.JsonWebTokens'          = ''
  'FluentValidation'                               = ''
  'FluentValidation.DependencyInjectionExtensions' = ''
  'Mapster'                                        = ''
  'Swashbuckle.AspNetCore.SwaggerUI'               = ''
  'Microsoft.NET.Test.Sdk'                         = ''
  'xunit.v3'                                       = ''
  'xunit.runner.visualstudio'                      = ''
  'NSubstitute'                                    = ''
  'Testcontainers.MsSql'                           = ''
}

$items = foreach ($id in $packages.Keys) {
  $prefix   = $packages[$id]
  $versions = (Invoke-RestMethod "https://api.nuget.org/v3-flatcontainer/$($id.ToLowerInvariant())/index.json").versions
  $latest   = $versions |
    Where-Object { $_ -notmatch '-' -and $_.StartsWith($prefix) } |
    Sort-Object { [version]$_ } |
    Select-Object -Last 1
  if (-not $latest) { throw "No stable version found for $id with prefix '$prefix'" }
  "    <PackageVersion Include=`"$id`" Version=`"$latest`" />"
}

$content = @"
<Project>
  <PropertyGroup>
    <ManagePackageVersionsCentrally>true</ManagePackageVersionsCentrally>
  </PropertyGroup>
  <ItemGroup>
$($items -join "`r`n")
  </ItemGroup>
</Project>
"@
Set-Content -Path Directory.Packages.props -Value $content -Encoding utf8

namespace Payroll.Api.ErrorHandling;

/// <summary>Converts C# property paths (e.g. "PayComponents[0].Amount") to the JSON paths the client sends.</summary>
public static class JsonPath
{
    public static string ToCamelCase(string propertyPath)
    {
        var segments = propertyPath.Split('.');
        for (var i = 0; i < segments.Length; i++)
        {
            var segment = segments[i];
            if (segment.Length > 0 && char.IsUpper(segment[0]))
            {
                segments[i] = char.ToLowerInvariant(segment[0]) + segment[1..];
            }
        }

        return string.Join('.', segments);
    }
}

```csharp
using FluentValidation;
using Microsoft.AspNetCore.Diagnostics;
using Microsoft.AspNetCore.Mvc;
using Payroll.Application.Common;
using Payroll.Domain.Common;

namespace Payroll.Api.ErrorHandling;

public sealed class ApiExceptionHandler(
    IProblemDetailsService problemDetailsService,
    ILogger<ApiExceptionHandler> logger) : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(
        HttpContext httpContext, Exception exception, CancellationToken cancellationToken)
    {
        var problem = ToProblem(exception);
        if (problem.Status == StatusCodes.Status500InternalServerError)
        {
            logger.LogError(exception, "Unhandled exception for {Method} {Path}",
                httpContext.Request.Method, httpContext.Request.Path);
        }

        httpContext.Response.StatusCode = problem.Status!.Value;
        return await problemDetailsService.TryWriteAsync(new ProblemDetailsContext
        {
            HttpContext = httpContext,
            ProblemDetails = problem,
            Exception = exception,
        });
    }

    private static ProblemDetails ToProblem(Exception exception) => exception switch
    {
        ValidationException validation => new HttpValidationProblemDetails(ToFieldErrors(validation))
        {
            Status = StatusCodes.Status400BadRequest,
            Title = "One or more validation errors occurred.",
        },
        NotFoundException notFound => Problem(StatusCodes.Status404NotFound, "Resource not found.", notFound.Message),
        ConflictException { Errors.Count: > 0 } conflict =>
            new HttpValidationProblemDetails(conflict.Errors.ToDictionary(e => e.Key, e => e.Value))
            {
                Status = StatusCodes.Status409Conflict,
                Title = "Conflict.",
                Detail = conflict.Message,
            },
        ConflictException conflict => Problem(StatusCodes.Status409Conflict, "Conflict.", conflict.Message),
        ForbiddenException forbidden => Problem(StatusCodes.Status403Forbidden, "Forbidden.", forbidden.Message),
        DomainException domain => Problem(StatusCodes.Status422UnprocessableEntity, "Business rule violated.", domain.Message),
        _ => Problem(StatusCodes.Status500InternalServerError, "An unexpected error occurred.", detail: null),
    };

    private static ProblemDetails Problem(int status, string title, string? detail) =>
        new() { Status = status, Title = title, Detail = detail };

    private static Dictionary<string, string[]> ToFieldErrors(ValidationException exception) =>
        exception.Errors
            .GroupBy(failure => JsonPath.ToCamelCase(failure.PropertyName))
            .ToDictionary(group => group.Key, group => group.Select(failure => failure.ErrorMessage).ToArray());
}
```

```csharp
namespace Payroll.Api.Configuration;

/// <summary>
/// Loads KEY=VALUE pairs from the git-ignored .env file into environment variables for local runs.
/// Variables that are already set (for example by docker compose) always win.
/// </summary>
public static class DotEnvFile
{
    public static void LoadFromAncestors(string startDirectory, string fileName = ".env")
    {
        var path = FindUpwards(startDirectory, fileName);
        if (path is null)
        {
            return;
        }

        foreach (var (key, value) in Parse(File.ReadLines(path)))
        {
            if (Environment.GetEnvironmentVariable(key) is null)
            {
                Environment.SetEnvironmentVariable(key, value);
            }
        }
    }

    public static string? FindUpwards(string startDirectory, string fileName)
    {
        for (var directory = new DirectoryInfo(startDirectory); directory is not null; directory = directory.Parent)
        {
            var candidate = Path.Combine(directory.FullName, fileName);
            if (File.Exists(candidate))
            {
                return candidate;
            }
        }

        return null;
    }

    public static IEnumerable<KeyValuePair<string, string>> Parse(IEnumerable<string> lines)
    {
        foreach (var rawLine in lines)
        {
            var line = rawLine.Trim();
            if (line.Length == 0 || line.StartsWith('#'))
            {
                continue;
            }

            var separator = line.IndexOf('=');
            if (separator <= 0)
            {
                continue;
            }

            var key = line[..separator].Trim();
            var value = Unquote(line[(separator + 1)..].Trim());
            yield return new KeyValuePair<string, string>(key, value);
        }
    }

    private static string Unquote(string value) =>
        value.Length >= 2 && (value[0] == '"' || value[0] == '\'') && value[^1] == value[0]
            ? value[1..^1]
            : value;
}
```
